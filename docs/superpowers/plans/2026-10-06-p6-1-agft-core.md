## Review Focus

- RC of a window must map a canonical plus P-site at `x` to a minus P-site at `S−1−x` and swap every plus/minus channel pair; applying RC twice must be the identity (Task 3).
- A shift must move sequence, targets and mask together and never wrap around the window edge (Task 3).
- The loss must be exactly 0 when prediction equals target (deviance/KL form), finite when a channel's target is all zero, and must ignore masked positions in both the total and the profile (Task 4).
- Reading the minus bigWig for canonical targets reads `[start+1, end+1)`; at a chromosome end that range is clipped and padded with zeros, not an error (Task 2).

```python
"""Canonical stranded target windows for fine-tuning (minus-strand RPF P-sites un-shifted)."""

from __future__ import annotations

from collections.abc import Sequence
from pathlib import Path

import numpy as np
import pyBigWig

from tisiago.seq2func.psite import RIBOTISH_MINUS_SHIFT
from tisiago.seq2func.samples import LIBRARIES
from tisiago.seq2func.tracks import track_prefix

STRAND_FILES = {"+": "plus", "-": "minus"}


def channel_layout(conditions: Sequence[str]) -> list[tuple[str, str, str]]:
    """Channels in model order: condition, then library, then (plus, minus)."""
    return [(c, lib, s) for c in conditions for lib in LIBRARIES for s in ("+", "-")]


def strand_pairs(layout: list[tuple[str, str, str]]) -> list[tuple[int, int]]:
    """Index pairs (plus, minus) for each condition/library."""
    index = {channel: i for i, channel in enumerate(layout)}
    return [(i, index[(c, lib, "-")]) for i, (c, lib, s) in enumerate(layout) if s == "+"]


def read_window(bw_path: str | Path, chrom: str, start: int, end: int) -> np.ndarray:
    """Per-base values for [start, end), zero-padded outside the chromosome."""
    out = np.zeros(end - start, dtype=np.float32)
    bw = pyBigWig.open(str(bw_path))
    try:
        size = bw.chroms().get(chrom, 0)
        lo, hi = max(start, 0), min(end, size)
        if hi > lo:
            values = np.asarray(bw.values(chrom, lo, hi), dtype=np.float32)
            out[lo - start : hi - start] = np.nan_to_num(values)
    finally:
        bw.close()
    return out


def load_targets(
    tracks_dir: str | Path,
    conditions: Sequence[str],
    chrom: str,
    start: int,
    end: int,
    rep: int | str = "pooled",
) -> np.ndarray:
    """Targets ``(end - start, C)`` in :func:`channel_layout` order, canonical P-sites."""
    columns = []
    for condition, library, strand in channel_layout(conditions):
        prefix = track_prefix(tracks_dir, condition, library, rep)
        shift = RIBOTISH_MINUS_SHIFT if strand == "-" and library != "rna" else 0
        path = f"{prefix}.{STRAND_FILES[strand]}.bw"
        columns.append(read_window(path, chrom, start + shift, end + shift))
    return np.stack(columns, axis=1)
```

```python
"""Strand-aware reverse-complement and shift augmentation for stranded track windows."""

from __future__ import annotations

import numpy as np


def reverse_complement(onehot, targets, mask, pairs):
    """Reverse-complement a window; swap each (plus, minus) channel pair."""
    order = np.arange(targets.shape[1])
    for plus, minus in pairs:
        order[plus], order[minus] = minus, plus
    return onehot[::-1, ::-1].copy(), targets[::-1][:, order].copy(), mask[::-1][:, order].copy()


def shift(onehot, targets, mask, offset: int):
    """Shift content by ``offset`` bases; vacated positions become N / zero / masked."""
    if offset == 0:
        return onehot.copy(), targets.copy(), mask.copy()
    out = (np.zeros_like(onehot), np.zeros_like(targets), np.zeros_like(mask))
    src = slice(0, -offset) if offset > 0 else slice(-offset, None)
    dst = slice(offset, None) if offset > 0 else slice(0, offset)
    for array, moved in zip((onehot, targets, mask), out, strict=True):
        moved[dst] = array[src]
    return out
```

"""Masked Poisson-multinomial loss in deviance form (0 at a perfect fit)."""

from __future__ import annotations

import jax.numpy as jnp

EPS = 1e-7


def _xlogy(x, y):
    return jnp.where(x > 0, x * jnp.log(jnp.maximum(y, EPS)), 0.0)


def poisson_multinomial(pred, target, mask, positional_weight: float = 5.0) -> dict:
    """Poisson deviance on window totals plus weighted multinomial KL on the profile."""
    pred = jnp.where(mask, pred.astype(jnp.float32), 0.0)
    target = jnp.where(mask, jnp.maximum(target.astype(jnp.float32), 0.0), 0.0)
    total_pred = pred.sum(axis=1)
    total_target = target.sum(axis=1)
    poisson = 2.0 * (
        total_pred - total_target + _xlogy(total_target, total_target) - _xlogy(total_target, total_pred)
    )
    p_target = target / jnp.maximum(total_target[:, None, :], EPS)
    p_pred = pred / jnp.maximum(total_pred[:, None, :], EPS)
    kl = (_xlogy(p_target, p_target) - _xlogy(p_target, p_pred)).sum(axis=1)
    poisson, kl = poisson.mean(), kl.mean()
    return {"loss": poisson + positional_weight * kl, "poisson": poisson, "multinomial": kl}
```

"""Per-condition loss mask: blank Ribo-seq over unexpressed genes is missing data, not zero."""

from __future__ import annotations

import numpy as np
import pandas as pd

MASKED_LIBRARIES = ("tis", "chx")


def expression_mask(layout, genes: pd.DataFrame, chrom: str, start: int, end: int) -> np.ndarray:
    """Boolean ``(end - start, C)`` mask; False = excluded from the loss."""
    mask = np.ones((end - start, len(layout)), dtype=bool)
    local = genes[(genes.chrom == chrom) & (genes.end > start) & (genes.start < end)]
    for index, (condition, library, strand) in enumerate(layout):
        if library not in MASKED_LIBRARIES:
            continue
        rows = local[(local.condition == condition) & (local.strand == strand)]
        silent = np.zeros(end - start, dtype=bool)
        for row in rows[~rows.expressed].itertuples():
            silent[max(row.start, start) - start : min(row.end, end) - start] = True
        for row in rows[rows.expressed].itertuples():
            silent[max(row.start, start) - start : min(row.end, end) - start] = False
        mask[:, index] = ~silent
    return mask
```
